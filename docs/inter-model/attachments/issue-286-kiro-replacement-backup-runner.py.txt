def run(argv: list[str], *, check: bool = True) -> subprocess.CompletedProcess[str]:
    return subprocess.run(argv, check=check, capture_output=True, text=True)


def require_authority_date(stage: str) -> str:
    """Bind both runner checks and governed date-tag generation to Chicago."""
    if os.environ.get("TZ") != AUTHORITY_TIMEZONE:
        raise RuntimeError(f"authority timezone changed before {stage}")
    time.tzset()
    chicago_date = datetime.now(AUTHORITY_ZONE).date().isoformat()
    effective_local_date = datetime.now().astimezone().date().isoformat()
    if chicago_date != RUN_DATE or effective_local_date != RUN_DATE:
        raise RuntimeError(
            f"authority date expired before {stage}: "
            f"Chicago={chicago_date}, effective={effective_local_date}"
        )
    return chicago_date


def require_backup_context_identity(ctx: BackupContext) -> None:
    """Require the reviewed profile, executable, roots, and repositories."""
    if ctx.profile is not BackupProfile.COMPLETE_DATA_V2:
        raise RuntimeError(f"backup profile mismatch: {ctx.profile.value}")
    if ctx.restic_bin.resolve() != RESTIC_BIN:
        raise RuntimeError(f"Restic executable mismatch: {ctx.restic_bin}")
    if sha256_file(ctx.restic_bin) != RESTIC_BIN_SHA256:
        raise RuntimeError("Restic executable digest mismatch")
    if ctx.data_root.resolve() != DATA_ROOT:
        raise RuntimeError("data root mismatch")
    if ctx.local_repository.locator != LOCAL_REPOSITORY:
        raise RuntimeError("local repository mismatch")
    if (
        ctx.external_repository is None
        or ctx.external_repository.locator != OFFSITE_REPOSITORY
    ):
        raise RuntimeError("offsite repository mismatch")


def require_offsite_mount_identity() -> dict[str, Any]:
    """Prove the repository is on the authorized mounted block device."""
    repository = Path(OFFSITE_REPOSITORY)
    source = run(
        ["findmnt", "-rn", "-T", str(repository), "-o", "SOURCE"]
    ).stdout.strip()
    target = run(
        ["findmnt", "-rn", "-T", str(repository), "-o", "TARGET"]
    ).stdout.strip()
    if not source or Path(source).resolve() != Path(OFFSITE_DEVICE).resolve():
        raise RuntimeError(f"offsite repository source mismatch: {source}")
    if target != str(OFFSITE_MOUNT):
        raise RuntimeError(f"offsite repository mount mismatch: {target}")
    mounted_uuid = run(["lsblk", "-no", "UUID", source]).stdout.strip()
    if mounted_uuid != OFFSITE_UUID:
        raise RuntimeError(f"offsite UUID mismatch after mount: {mounted_uuid}")
    if repository.resolve() != repository or not repository.is_dir():
        raise RuntimeError("offsite repository path is missing or redirected")
    repository_device = os.stat(repository).st_dev
    block_device = os.stat(OFFSITE_DEVICE).st_rdev
    if repository_device != block_device:
        raise RuntimeError(
            "offsite repository is not resident on the authorized device"
        )
    return {
        "source": source,
        "target": target,
        "uuid": mounted_uuid,
        "repository_st_dev": repository_device,
        "block_device_st_rdev": block_device,
    }


def unit_state(unit: str, verb: str) -> str:
    proc = run(["systemctl", "--user", verb, unit], check=False)
    return (proc.stdout or proc.stderr).strip()


def require_service_quiescent(service: str) -> dict[str, str]:
    active = unit_state(service, "is-active")
    main_pid = run(
        ["systemctl", "--user", "show", "-p", "MainPID", "--value", service]
    ).stdout.strip()
    if active not in {"inactive", "failed"} or main_pid != "0":
        raise RuntimeError(
            f"service did not quiesce: {service} ({active}, MainPID={main_pid})"
        )
    return {"active": active, "main_pid": main_pid}

class WriterCensusRefusal(RuntimeError):
    """Refusal carrying bounded process evidence for post-exit attribution."""

    def __init__(self, evidence: dict[str, Any]):
        self.evidence = evidence
        super().__init__(
            "writer census refusal: "
            f"classifier={evidence['classifier']} pid={evidence['pid']} "
            f"cmdline_sha256={evidence['cmdline_sha256']}"
        )


def convmem_writer_match(argv: list[str]) -> dict[str, Any] | None:
    """Identify launcher, Python-script, and ``python -m`` CLI forms."""
    for index, argument in enumerate(argv):
        if Path(argument).name not in {"convmem", "convmem.py"}:
            continue
        for candidate in argv[index + 1 :]:
            if candidate in CONVMEM_WRITE_COMMANDS:
                return {
                    "classifier": "convmem_write_command",
                    "launcher": Path(argument).name,
                    "launcher_index": index,
                    "subcommand": candidate,
                }
    return None


def refuse_writer(
    entry: Path,
    *,
    raw_cmdline: bytes,
    argv: list[str],
    cwd: Path | None,
    match: dict[str, Any],
) -> None:
    """Raise with stable, bounded evidence even if the process exits next."""
    stat_fields: list[str] = []
    try:
        stat_fields = (
            (entry / "stat")
            .read_text(encoding="utf-8")
            .rsplit(")", 1)[1]
            .split()
        )
    except (FileNotFoundError, PermissionError, IndexError):
        pass
    try:
        executable = str((entry / "exe").resolve())
    except (FileNotFoundError, PermissionError):
        executable = None
    try:
        uid = entry.stat().st_uid
    except (FileNotFoundError, PermissionError):
        uid = None
    evidence: dict[str, Any] = {
        **match,
        "pid": int(entry.name),
        "ppid": int(stat_fields[1]) if len(stat_fields) > 1 else None,
        "start_time_ticks": int(stat_fields[19]) if len(stat_fields) > 19 else None,
        "uid": uid,
        "cwd": str(cwd) if cwd is not None else None,
        "executable": executable,
        "argv_count": len(argv),
        "cmdline_sha256": hashlib.sha256(raw_cmdline).hexdigest(),
    }
    raise WriterCensusRefusal(evidence)


def assert_no_uncontrolled_writer() -> None:
    current_pid = os.getpid()
    for entry in Path("/proc").iterdir():
        if not entry.name.isdigit() or int(entry.name) == current_pid:
            continue
        try:
            raw_cmdline = (entry / "cmdline").read_bytes()
            argv = [
                item.decode("utf-8", "replace")
                for item in raw_cmdline.split(b"\0")
                if item
            ]
            command = " ".join(argv)
        except (FileNotFoundError, PermissionError):
            continue
        try:
            cwd = (entry / "cwd").resolve()
        except (FileNotFoundError, PermissionError):
            cwd = None
        if "test_index_loud_failures.py" in command:
            refuse_writer(
                entry,
                raw_cmdline=raw_cmdline,
                argv=argv,
                cwd=cwd,
                match={
                    "classifier": "stale_failure_test",
                    "matched_marker": "test_index_loud_failures.py",
                },
            )
        if any(Path(item).name.startswith("pytest") for item in argv) and (
            (cwd is not None and "convmem" in str(cwd)) or "convmem" in command
        ):
            refuse_writer(
                entry,
                raw_cmdline=raw_cmdline,
                argv=argv,
                cwd=cwd,
                match={"classifier": "convmem_pytest"},
            )
        convmem_match = convmem_writer_match(argv)
        if convmem_match is not None:
            refuse_writer(
                entry,
                raw_cmdline=raw_cmdline,
                argv=argv,
                cwd=cwd,
                match=convmem_match,
            )
        if "convmem-watch" in command:
            refuse_writer(
                entry,
                raw_cmdline=raw_cmdline,
                argv=argv,
                cwd=cwd,
                match={
                    "classifier": "convmem_watcher",
                    "matched_marker": "convmem-watch",
                },
            )
        direct_marker = next(
            (marker for marker in DIRECT_WRITER_MARKERS if marker in command), None
        )
        if direct_marker is not None:
            refuse_writer(
                entry,
                raw_cmdline=raw_cmdline,
                argv=argv,
                cwd=cwd,
                match={
                    "classifier": "direct_writer",
                    "matched_marker": direct_marker,
                },
            )
        if any(Path(item).name == "restic" for item in argv) and (
            LOCAL_REPOSITORY in command or OFFSITE_REPOSITORY in command
        ):
            refuse_writer(
                entry,
                raw_cmdline=raw_cmdline,
                argv=argv,
                cwd=cwd,
                match={"classifier": "governed_restic_operation"},
            )


def reserve_result() -> int:
    """Create the one-shot marker before any timer, mount, or corpus mutation."""
    if os.path.lexists(RESULT):
        raise FileExistsError(f"refusing existing result or symlink: {RESULT}")
    flags = os.O_WRONLY | os.O_CREAT | os.O_EXCL
    if hasattr(os, "O_NOFOLLOW"):
        flags |= os.O_NOFOLLOW
    fd = os.open(RESULT, flags, 0o600)
    try:
        os.fchmod(fd, 0o600)
    except Exception:
        os.close(fd)
        raise
    return fd


def write_reserved_result(fd: int, payload: dict[str, Any]) -> None:
    encoded = (json.dumps(payload, sort_keys=True, separators=(",", ":")) + "\n").encode()
    os.lseek(fd, 0, os.SEEK_SET)
    os.ftruncate(fd, 0)
    remaining = memoryview(encoded)
    while remaining:
        written = os.write(fd, remaining)
        if written <= 0:
            raise OSError("failed to write reserved result")
        remaining = remaining[written:]
    os.fsync(fd)


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--execute", action="store_true")
    args = parser.parse_args()
    if not args.execute:
        raise SystemExit("refusing without --execute")

    # This is both the evidence destination and the durable one-shot marker.
    # Reserve it before stopping timers, mounting, or touching the corpus.
    result_fd = reserve_result()
    try:
        write_reserved_result(
            result_fd,
            {
                "status": "RESERVED",
                "runtime_revision": RUNTIME_REVISION,
                "expected_inventory_sha256": EXPECTED_INVENTORY,
            },
        )
    except Exception:
        os.close(result_fd)
        raise

    result: dict[str, Any] = {
        "status": "FAIL",
        "runtime_revision": RUNTIME_REVISION,
        "expected_inventory_sha256": EXPECTED_INVENTORY,
        "local_snapshot_id": None,
        "offsite_snapshot_id": None,
        "watcher_before": {},
        "cleanup": {},
    }
    timer_before: dict[str, dict[str, str]] = {}
    watcher_before: dict[str, str] = {}
    mount_command_succeeded = False
    mount_identity: dict[str, Any] | None = None
    operation_error: Exception | None = None
    result_write_error: Exception | None = None
    cleanup_errors: list[str] = []
    received_signals: list[str] = []
    in_cleanup = False

    def handle_signal(signum: int, _frame: object) -> None:
        name = signal.Signals(signum).name
        received_signals.append(name)
        if not in_cleanup:
            raise RuntimeError(f"received termination signal: {name}")

    handled_signals = (
        signal.SIGHUP,
        signal.SIGINT,
        signal.SIGQUIT,
        signal.SIGTERM,
        signal.SIGALRM,
        signal.SIGUSR1,
        signal.SIGUSR2,
    )
    previous_handlers = {
        signum: signal.getsignal(signum) for signum in handled_signals
    }
    for signum in handled_signals:
        signal.signal(signum, handle_signal)

    try:
        timer_before = {
            timer: {
                "active": unit_state(timer, "is-active"),
                "enabled": unit_state(timer, "is-enabled"),
            }
            for timer in TIMERS
        }
        watcher_before = {
            "active": unit_state("convmem-watch.service", "is-active"),
            "enabled": unit_state("convmem-watch.service", "is-enabled"),
        }
        result["watcher_before"] = watcher_before
        runtime_head = run(["git", "-C", str(RUNTIME), "rev-parse", "HEAD"])
        if runtime_head.stdout.strip() != RUNTIME_REVISION:
            raise RuntimeError("runtime revision mismatch")
        runtime_status = run(
            [
                "git",
                "-C",
                str(RUNTIME),
                "status",
                "--porcelain",
                "--untracked-files=all",
            ]
        )
        if runtime_status.stdout:
            raise RuntimeError("runtime worktree is not clean")
        if sha256_file(CONFIG) != CONFIG_SHA256:
            raise RuntimeError("config digest mismatch")
        if sha256_file(SOURCE) != SOURCE_SHA256:
            raise RuntimeError("source digest mismatch")
        if source_state_id(str(SOURCE)) != SOURCE_STATE_ID:
            raise RuntimeError("source identity mismatch")
        result["authority_date"] = require_authority_date("preflight")
        result["authority_timezone"] = AUTHORITY_TIMEZONE
        if watcher_before["active"] != "inactive":
            raise RuntimeError("watcher is not inactive")
        if watcher_before["enabled"] != "disabled":
            raise RuntimeError("watcher is not disabled")
        main_pid = run(
            [
                "systemctl",
                "--user",
                "show",
                "-p",
                "MainPID",
                "--value",
                "convmem-watch.service",
            ]
        ).stdout.strip()
        if main_pid != "0":
            raise RuntimeError(f"watcher MainPID is {main_pid}")

        for timer in TIMERS:
            run(["systemctl", "--user", "stop", timer])
        result["service_quiescence"] = {
            service: require_service_quiescent(service) for service in SERVICES
        }
        assert_no_uncontrolled_writer()

        preexisting_mount = run(
            ["findmnt", "-rn", "-S", OFFSITE_DEVICE, "-o", "TARGET"],
            check=False,
        ).stdout.strip()
        if preexisting_mount:
            raise RuntimeError(f"offsite device already mounted: {preexisting_mount}")
        uuid = run(["lsblk", "-no", "UUID", OFFSITE_DEVICE]).stdout.strip()
        if uuid != OFFSITE_UUID:
            raise RuntimeError(f"offsite UUID mismatch before mount: {uuid}")
        run(["udisksctl", "mount", "-b", OFFSITE_DEVICE])
        mount_command_succeeded = True
        mount_identity = require_offsite_mount_identity()
        result["offsite_mount_identity"] = mount_identity
        offsite_repository_path = Path(OFFSITE_REPOSITORY)
        if not offsite_repository_path.is_dir() or not os.access(
            offsite_repository_path, os.R_OK | os.W_OK
        ):
            raise RuntimeError("offsite repository is not readable and writable")
        offsite_fs = os.statvfs(offsite_repository_path)
        offsite_free_bytes = offsite_fs.f_bavail * offsite_fs.f_frsize
        if offsite_free_bytes <= 0:
            raise RuntimeError("offsite repository has no available space")
        result["offsite_free_bytes_before"] = offsite_free_bytes

        ctx = BackupContext.from_env_file(Path("/home/lauer/.config/convmem/restic.env"))
        require_backup_context_identity(ctx)
        result["backup_profile"] = ctx.profile.value
        result["restic_bin"] = str(ctx.restic_bin)
        result["restic_bin_sha256"] = RESTIC_BIN_SHA256

        # Read-only offsite readiness before the first write.
        require_backup_context_identity(ctx)
        require_offsite_mount_identity()
        old_source = resolve_snapshot(ctx, requested_id=OLD_LOCAL)
        old_destination = resolve_copy_destination(ctx, old_source)
        if old_destination.id != OLD_OFFSITE:
            raise RuntimeError("offsite readiness lineage mismatch")

        cfg = load_config(CONFIG)
        with exclusive_writer_lease(lock_path=LOCK, timeout_ms=120_000):
            assert_no_uncontrolled_writer()
            pre_inventory, pre_digest = current_inventory(cfg)
            if pre_digest != EXPECTED_INVENTORY:
                raise RuntimeError(f"mutation inventory mismatch: {pre_digest}")

            require_authority_date("backup evidence")
            backup_evidence = capture_backup_evidence(DATA_ROOT)
            require_authority_date("local snapshot")
            require_backup_context_identity(ctx)
            source, backup_argv = backup_data_root(
                ctx, extra_tags=("kiro-successor-preflight",)
            )
            result["local_snapshot_id"] = source.id
            source = resolve_snapshot(ctx, requested_id=source.id)
            if set(source.tags) != REQUIRED_TAGS:
                raise RuntimeError(f"local snapshot tags mismatch: {sorted(source.tags)}")
            if source.paths != (str(DATA_ROOT),):
                raise RuntimeError(f"local snapshot paths mismatch: {source.paths}")

            post_local_inventory, post_local_digest = current_inventory(cfg)
            if post_local_inventory != pre_inventory or post_local_digest != pre_digest:
                raise RuntimeError("mutation inventory changed during local snapshot")

            # Full-ID handoff: the ID returned by backup_data_root is passed
            # directly into the governed copy primitive. No current/latest
            # reselection occurs between creation and copy.
            require_authority_date("offsite copy")
            require_backup_context_identity(ctx)
            mount_identity = require_offsite_mount_identity()
            result["offsite_mount_identity_before_copy"] = mount_identity
            copy_argv = copy_snapshot_to_external(ctx, source.id)
            destination = resolve_copy_destination(ctx, source)
            result["offsite_snapshot_id"] = destination.id
            if destination.original != source.id:
                raise RuntimeError("offsite original does not equal created local ID")
            if destination.tags != source.tags:
                raise RuntimeError("offsite tags do not equal created local tags")

            assert_no_uncontrolled_writer()
            final_inventory, final_digest = current_inventory(cfg)
            if final_inventory != pre_inventory or final_digest != pre_digest:
                raise RuntimeError("mutation inventory changed before final evidence")
            require_authority_date("final evidence")

            result.update(
                {
                    "status": "PASS",
                    "inventory_sha256": pre_digest,
                    "post_local_inventory_sha256": post_local_digest,
                    "final_inventory_sha256": final_digest,
                    "tree": source.tree,
                    "time": source.time.isoformat(),
                    "paths": list(source.paths),
                    "tags": sorted(source.tags),
                    "offsite_original": destination.original,
                    "backup_argv": list(backup_argv),
                    "copy_argv": list(copy_argv),
                    "backup_evidence_schema": backup_evidence.get(
                        "evidence_schema_version"
                    ),
                    "backup_evidence_captured_at": backup_evidence.get(
                        "captured_at"
                    ),
                    "backup_evidence_sha256": sha256_file(BACKUP_EVIDENCE),
                }
            )
    except Exception as exc:  # noqa: BLE001 - preserve evidence, then fail closed
        operation_error = exc
        result["status"] = "FAIL"
        result["error_type"] = type(exc).__name__
        result["error"] = str(exc)
        if isinstance(exc, WriterCensusRefusal):
            result["writer_census_refusal"] = exc.evidence
    finally:
        in_cleanup = True
        if mount_command_succeeded:
            try:
                result["offsite_mount_identity_before_unmount"] = (
                    require_offsite_mount_identity()
                )
            except Exception as exc:  # noqa: BLE001 - refuse wrong-device cleanup
                cleanup_errors.append(f"offsite identity blocks unmount: {exc}")
            else:
                proc = run(
                    ["udisksctl", "unmount", "-b", OFFSITE_DEVICE], check=False
                )
                if proc.returncode != 0:
                    cleanup_errors.append("offsite unmount failed")
        for timer, before in timer_before.items():
            if before["active"] == "active":
                proc = run(["systemctl", "--user", "start", timer], check=False)
                if proc.returncode != 0:
                    cleanup_errors.append(f"timer restore failed: {timer}")
        timer_after = {
            timer: {
                "active": unit_state(timer, "is-active"),
                "enabled": unit_state(timer, "is-enabled"),
            }
            for timer in TIMERS
        }
        mounted_after = bool(
            run(
                ["findmnt", "-rn", "-S", OFFSITE_DEVICE, "-o", "TARGET"],
                check=False,
            ).stdout.strip()
        )
        watcher_after = {
            "active": unit_state("convmem-watch.service", "is-active"),
            "enabled": unit_state("convmem-watch.service", "is-enabled"),
        }
        result["cleanup"] = {
            "timer_before": timer_before,
            "timer_after": timer_after,
            "offsite_mounted_after": mounted_after,
            "watcher_after": watcher_after,
            "errors": cleanup_errors,
        }
        if (
            cleanup_errors
            or timer_after != timer_before
            or mounted_after
            or watcher_after != result["watcher_before"]
        ):
            result["status"] = "FAIL"
            result["cleanup_error"] = "cleanup or state restoration mismatch"
        if received_signals:
            result["status"] = "FAIL"
            result["received_signals"] = received_signals
        try:
            write_reserved_result(result_fd, result)
        except Exception as exc:  # noqa: BLE001 - marker remains reserved
            result_write_error = exc
            result["status"] = "FAIL"
            result["result_write_error"] = f"{type(exc).__name__}: {exc}"
        finally:
            os.close(result_fd)
            for signum, previous in previous_handlers.items():
                signal.signal(signum, previous)

    print(json.dumps(result, sort_keys=True))
    return (
        0
        if result["status"] == "PASS"
        and operation_error is None
        and result_write_error is None
        else 1
    )


if __name__ == "__main__":
    raise SystemExit(main())
